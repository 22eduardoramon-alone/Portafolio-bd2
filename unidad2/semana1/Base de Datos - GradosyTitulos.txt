IF OBJECT_ID('Academico.Facultad', 'U') IS NULL
BEGIN
    CREATE TABLE Academico.Facultad (
        IdFacultad TINYINT IDENTITY(1,1) NOT NULL,
        NombreFacultad VARCHAR(100) NOT NULL,
        NombreDecano VARCHAR(150) NULL,
        Activo BIT NOT NULL
            CONSTRAINT DF_Facultad_Activo DEFAULT 1,

        CONSTRAINT PK_Facultad
            PRIMARY KEY (IdFacultad),

        CONSTRAINT UQ_Facultad_Nombre
            UNIQUE (NombreFacultad)
    );
END;
GO

IF OBJECT_ID('Academico.ProgramaEstudios', 'U') IS NULL
BEGIN
    CREATE TABLE Academico.ProgramaEstudios (
        IdPrograma SMALLINT IDENTITY(1,1) NOT NULL,
        IdFacultad TINYINT NOT NULL,
        NombrePrograma VARCHAR(150) NOT NULL,
        NumSemestres TINYINT NOT NULL,
        Modalidad VARCHAR(15) NOT NULL,
        Activo BIT NOT NULL
            CONSTRAINT DF_Programa_Activo DEFAULT 1,

        CONSTRAINT PK_ProgramaEstudios
            PRIMARY KEY (IdPrograma),

        CONSTRAINT FK_Programa_Facultad
            FOREIGN KEY (IdFacultad)
            REFERENCES Academico.Facultad(IdFacultad),

        CONSTRAINT CHK_Programa_Semestres
            CHECK (NumSemestres IN (10, 12, 14)),

        CONSTRAINT CHK_Programa_Modalidad
            CHECK (Modalidad IN ('PRESENCIAL', 'SEMIPRESENCIAL')),

        CONSTRAINT UQ_Programa_Nombre_Modalidad
            UNIQUE (NombrePrograma, Modalidad)
    );
END;
GO

IF OBJECT_ID('Seguridad.Rol', 'U') IS NULL
BEGIN
    CREATE TABLE Seguridad.Rol (
        IdRol SMALLINT IDENTITY(1,1) NOT NULL,
        NombreRol VARCHAR(50) NOT NULL,
        Descripcion VARCHAR(250) NULL,
        EsSistema BIT NOT NULL
            CONSTRAINT DF_Rol_EsSistema DEFAULT 0,
        Activo BIT NOT NULL
            CONSTRAINT DF_Rol_Activo DEFAULT 1,
        FechaCreacion DATETIME2 NOT NULL
            CONSTRAINT DF_Rol_Fecha DEFAULT SYSDATETIME(),

        CONSTRAINT PK_Rol PRIMARY KEY (IdRol),
        CONSTRAINT UQ_Rol_Nombre UNIQUE (NombreRol)
    );
END;
GO

IF OBJECT_ID('Seguridad.Permiso', 'U') IS NULL
BEGIN
    CREATE TABLE Seguridad.Permiso (
        IdPermiso INT IDENTITY(1,1) NOT NULL,
        NombrePermiso VARCHAR(100) NOT NULL,

        CONSTRAINT PK_Permiso PRIMARY KEY (IdPermiso),
        CONSTRAINT UQ_Permiso_Nombre UNIQUE (NombrePermiso)
    );
END;
GO

IF OBJECT_ID('Seguridad.RolPermiso', 'U') IS NULL
BEGIN
    CREATE TABLE Seguridad.RolPermiso (
        IdRol SMALLINT NOT NULL,
        IdPermiso INT NOT NULL,
        FechaAsignacion DATETIME2 NOT NULL
            CONSTRAINT DF_RolPermiso_Fecha DEFAULT SYSDATETIME(),

        CONSTRAINT PK_RolPermiso
            PRIMARY KEY (IdRol, IdPermiso),

        CONSTRAINT FK_RolPermiso_Rol
            FOREIGN KEY (IdRol)
            REFERENCES Seguridad.Rol(IdRol),

        CONSTRAINT FK_RolPermiso_Permiso
            FOREIGN KEY (IdPermiso)
            REFERENCES Seguridad.Permiso(IdPermiso)
    );
END;
GO

-- Completar los programas hasta alcanzar 30.
;WITH Numeros AS (
    SELECT TOP (30)
        ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS N
    FROM sys.all_objects A
    CROSS JOIN sys.all_objects B
)
INSERT INTO Academico.ProgramaEstudios
    (IdFacultad, NombrePrograma, NumSemestres, Modalidad)
SELECT
    F.IdFacultad,
    CONCAT('Programa de Prueba ', RIGHT('00' + CAST(N AS VARCHAR(2)), 2)),
    CASE N % 3 WHEN 0 THEN 14 WHEN 1 THEN 10 ELSE 12 END,
    CASE WHEN N % 2 = 0
         THEN 'PRESENCIAL'
         ELSE 'SEMIPRESENCIAL'
    END
FROM Numeros N
CROSS JOIN (
    SELECT TOP (1) IdFacultad
    FROM Academico.Facultad
    ORDER BY IdFacultad
) F
WHERE NOT EXISTS (
    SELECT 1
    FROM Academico.ProgramaEstudios P
    WHERE P.NombrePrograma =
        CONCAT('Programa de Prueba ', RIGHT('00' + CAST(N.N AS VARCHAR(2)), 2))
)
AND (SELECT COUNT(*) FROM Academico.ProgramaEstudios) < 30;
GO

-- 8. INSERTAR 30 RELACIONES ROL-PERMISO
-- Se asigna un permiso diferente a cada rol cuando es posible.
-- ============================================================

;WITH RolesNumerados AS (
    SELECT
        IdRol,
        ROW_NUMBER() OVER (ORDER BY IdRol) AS N
    FROM Seguridad.Rol
),
PermisosNumerados AS (
    SELECT
        IdPermiso,
        ROW_NUMBER() OVER (ORDER BY IdPermiso) AS N
    FROM Seguridad.Permiso
)
INSERT INTO Seguridad.RolPermiso (IdRol, IdPermiso)
SELECT R.IdRol, P.IdPermiso
FROM RolesNumerados R
INNER JOIN PermisosNumerados P
    ON P.N = R.N
WHERE R.N <= 30
AND NOT EXISTS (
    SELECT 1
    FROM Seguridad.RolPermiso RP
    WHERE RP.IdRol = R.IdRol
      AND RP.IdPermiso = P.IdPermiso
);
GO
