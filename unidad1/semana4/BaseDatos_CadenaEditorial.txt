-- ==============================================================================
-- 2. CREACIÓN DE TABLAS
-- ==============================================================================

CREATE TABLE Sucursal (
    CodigoSucursal VARCHAR(10)  NOT NULL,
    Domicilio      VARCHAR(150) NOT NULL,
    Telefono       VARCHAR(20)  NOT NULL,
    CONSTRAINT PK_Sucursal PRIMARY KEY (CodigoSucursal)
);
GO

CREATE TABLE Empleado (
    NIF            VARCHAR(15) NOT NULL,
    Nombre         VARCHAR(50) NOT NULL,
    Apellidos      VARCHAR(50) NOT NULL,
    Telefono       VARCHAR(20) NOT NULL,
    CodigoSucursal VARCHAR(10) NOT NULL,
    CONSTRAINT PK_Empleado PRIMARY KEY (NIF),
    CONSTRAINT FK_Empleado_Sucursal FOREIGN KEY (CodigoSucursal)
        REFERENCES Sucursal(CodigoSucursal)
);
GO

CREATE TABLE Revista (
    NumeroRegistro VARCHAR(20)  NOT NULL,
    Titulo         VARCHAR(100) NOT NULL,
    Periodicidad   VARCHAR(50)  NOT NULL,
    Tipo           VARCHAR(50)  NOT NULL,
    CONSTRAINT PK_Revista PRIMARY KEY (NumeroRegistro)
);
GO

CREATE TABLE Sucursal_Revista (
    CodigoSucursal VARCHAR(10) NOT NULL,
    NumeroRegistro VARCHAR(20) NOT NULL,
    CONSTRAINT PK_Sucursal_Revista PRIMARY KEY (CodigoSucursal, NumeroRegistro),
    CONSTRAINT FK_SucRev_Sucursal FOREIGN KEY (CodigoSucursal)
        REFERENCES Sucursal(CodigoSucursal),
    CONSTRAINT FK_SucRev_Revista FOREIGN KEY (NumeroRegistro)
        REFERENCES Revista(NumeroRegistro)
);
GO

CREATE TABLE Periodista (
    NIF          VARCHAR(15)  NOT NULL,
    Nombre       VARCHAR(50)  NOT NULL,
    Apellidos    VARCHAR(50)  NOT NULL,
    Telefono     VARCHAR(20)  NOT NULL,
    Especialidad VARCHAR(100) NOT NULL,
    CONSTRAINT PK_Periodista PRIMARY KEY (NIF)
);
GO

CREATE TABLE Periodista_Revista (
    NIF_Periodista VARCHAR(15) NOT NULL,
    NumeroRegistro VARCHAR(20) NOT NULL,
    CONSTRAINT PK_Periodista_Revista PRIMARY KEY (NIF_Periodista, NumeroRegistro),
    CONSTRAINT FK_PerRev_Periodista FOREIGN KEY (NIF_Periodista)
        REFERENCES Periodista(NIF),
    CONSTRAINT FK_PerRev_Revista FOREIGN KEY (NumeroRegistro)
        REFERENCES Revista(NumeroRegistro)
);
GO

-- Tabla nueva: artículos escritos por un periodista para una revista
CREATE TABLE Articulo (
    IdArticulo       INT IDENTITY(1,1) NOT NULL,
    Titulo           VARCHAR(200) NOT NULL,
    FechaPublicacion DATE         NOT NULL,
    NIF_Periodista   VARCHAR(15)  NOT NULL,
    NumeroRegistro   VARCHAR(20)  NOT NULL,
    CONSTRAINT PK_Articulo PRIMARY KEY (IdArticulo),
    CONSTRAINT FK_Articulo_Periodista FOREIGN KEY (NIF_Periodista)
        REFERENCES Periodista(NIF),
    CONSTRAINT FK_Articulo_Revista FOREIGN KEY (NumeroRegistro)
        REFERENCES Revista(NumeroRegistro)
);
GO

CREATE TABLE Seccion (
    NumeroRegistro VARCHAR(20)  NOT NULL,
    Titulo         VARCHAR(100) NOT NULL,
    Extension      INT          NOT NULL,
    CONSTRAINT PK_Seccion PRIMARY KEY (NumeroRegistro, Titulo),
    CONSTRAINT FK_Seccion_Revista FOREIGN KEY (NumeroRegistro)
        REFERENCES Revista(NumeroRegistro),
    CONSTRAINT CK_Seccion_Extension CHECK (Extension > 0)
);
GO

CREATE TABLE Ejemplar (
    NumeroRegistro     VARCHAR(20) NOT NULL,
    Fecha              DATE        NOT NULL,
    NumeroPaginas      INT         NOT NULL,
    EjemplaresVendidos INT         NOT NULL,
    CONSTRAINT PK_Ejemplar PRIMARY KEY (NumeroRegistro, Fecha),
    CONSTRAINT FK_Ejemplar_Revista FOREIGN KEY (NumeroRegistro)
        REFERENCES Revista(NumeroRegistro),
    CONSTRAINT CK_Ejemplar_Paginas CHECK (NumeroPaginas > 0),
    CONSTRAINT CK_Ejemplar_Vendidos CHECK (EjemplaresVendidos >= 0)
);
GO

-- ==============================================================================
-- 3. ÍNDICES SOBRE CLAVES FORÁNEAS
-- ==============================================================================
CREATE INDEX IX_Empleado_Sucursal  ON Empleado(CodigoSucursal);
CREATE INDEX IX_SucRev_Revista     ON Sucursal_Revista(NumeroRegistro);
CREATE INDEX IX_PerRev_Revista     ON Periodista_Revista(NumeroRegistro);
CREATE INDEX IX_Articulo_Periodista ON Articulo(NIF_Periodista);
CREATE INDEX IX_Articulo_Revista    ON Articulo(NumeroRegistro);
GO

-- ==============================================================================
-- 5. CONSULTAS DE VERIFICACIÓN
-- ==============================================================================

-- Conteo de registros por tabla (esperado: 20 en todas)
SELECT 'Sucursal' AS Tabla, COUNT(*) AS Total FROM Sucursal
UNION ALL SELECT 'Empleado', COUNT(*) FROM Empleado
UNION ALL SELECT 'Revista', COUNT(*) FROM Revista
UNION ALL SELECT 'Sucursal_Revista', COUNT(*) FROM Sucursal_Revista
UNION ALL SELECT 'Periodista', COUNT(*) FROM Periodista
UNION ALL SELECT 'Articulo', COUNT(*) FROM Articulo
UNION ALL SELECT 'Periodista_Revista', COUNT(*) FROM Periodista_Revista;

-- Revistas publicadas por cada sucursal
SELECT s.CodigoSucursal, s.Domicilio, r.Titulo
FROM Sucursal s
JOIN Sucursal_Revista sr ON s.CodigoSucursal = sr.CodigoSucursal
JOIN Revista r           ON r.NumeroRegistro = sr.NumeroRegistro
ORDER BY s.CodigoSucursal;

-- Artículos con su periodista y su revista
SELECT a.Titulo AS Articulo, a.FechaPublicacion,
       p.Nombre + ' ' + p.Apellidos AS Periodista,
       r.Titulo AS Revista
FROM Articulo a
JOIN Periodista p ON p.NIF = a.NIF_Periodista
JOIN Revista r    ON r.NumeroRegistro = a.NumeroRegistro
ORDER BY a.FechaPublicacion;
GO
